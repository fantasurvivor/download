rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn() && get(/databases/$(db)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
    function validInviteCode(c) { return c is string && c.matches('^[A-Z0-9]{6,20}$'); }
    // Iscrizione con PayPal (solo partite private): prezzo in euro e link paypal.me dell'organizzatore.
    function validPaypal(g) {
      return g.get('paypalEnabled', false) == false
        || (g.paypalEnabled == true && g.isPublic == false
            && g.paypalPrice is number && g.paypalPrice > 0 && g.paypalPrice <= 100
            && g.paypalLink is string && g.paypalLink.matches('^https://(www[.])?paypal[.]me/[A-Za-z0-9_.-]{1,40}$'));
    }
    // Iscrizione con buono regalo Amazon (solo partite private): importo in euro.
    function validAmazon(g) {
      return g.get('amazonEnabled', false) == false
        || (g.amazonEnabled == true && g.isPublic == false
            && g.amazonAmount is number && g.amazonAmount > 0 && g.amazonAmount <= 100);
    }
    function validPayments(g) { return validPaypal(g) && validAmazon(g); }
    function validUsername(n) { return n is string && n.matches('^[a-z0-9_.]{3,20}$'); }
    // Foto profilo: JPEG in base64 (~100 KB) salvata nel documento (niente Storage sul piano Spark).
    function validPhoto(p) { return p == null || (p is string && p.size() <= 140000); }
    function usernameOwned(n) {
      return getAfter(/databases/$(db)/documents/usernames/$(n)).data.uid == request.auth.uid;
    }

    // Profilo pubblico: niente dati di contatto. Il ruolo admin si imposta solo dalla Console.
    match /users/{uid} {
      allow read: if signedIn();
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['displayName', 'role', 'createdAt', 'avatar'])
        && request.resource.data.role == 'user';
      allow update: if signedIn() && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['displayName', 'username', 'avatar', 'photo'])
        && validPhoto(request.resource.data.get('photo', null))
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['username'])
          || (validUsername(request.resource.data.username)
              && usernameOwned(request.resource.data.username)
              // il vecchio username va liberato nella stessa transazione
              && (resource.data.get('username', null) == null
                  || !existsAfter(/databases/$(db)/documents/usernames/$(resource.data.username)))));

      // Dati privati (email, token notifiche): li scrive solo il proprietario; l'admin può leggerli.
      match /private/{doc} {
        allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
        allow write: if signedIn() && request.auth.uid == uid;
      }
    }

    // Username unici (minuscoli): usernames/{nome} = { uid }.
    match /usernames/{name} {
      allow get: if signedIn();
      allow list: if false;
      allow create: if signedIn() && validUsername(name)
        && request.resource.data.keys().hasOnly(['uid']) && request.resource.data.uid == request.auth.uid
        && getAfter(/databases/$(db)/documents/users/$(request.auth.uid)).data.username == name;
      allow delete: if signedIn() && resource.data.uid == request.auth.uid
        && getAfter(/databases/$(db)/documents/users/$(request.auth.uid)).data.get('username', '') != name;
      allow update: if false;
    }

    // Albo d'oro: tutti leggono, solo l'admin scrive.
    match /hallOfFame/{id} {
      allow read: if signedIn();
      allow delete: if isAdmin();
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly(['tournamentName', 'gameId', 'winnerUid', 'winnerUsername',
             'winnerPhoto', 'winnerAvatar', 'winnerDisplayName', 'prize', 'date', 'createdAt'])
        && request.resource.data.tournamentName is string && request.resource.data.tournamentName.size() > 0
        && request.resource.data.winnerUid is string
        && request.resource.data.date is timestamp
        && validPhoto(request.resource.data.get('winnerPhoto', null));
    }

    // Dati Serie A: scritti solo dallo script GitHub Actions (Admin SDK, ignora le regole).
    match /matchdays/{md} { allow read: if signedIn(); allow write: if false; }
    match /teams/{id}     { allow read: if signedIn(); allow write: if false; }
    match /meta/{id}      { allow read: if signedIn(); allow write: if false; }

    // Codici invito: si leggono solo conoscendo il codice (get), mai elencati.
    match /inviteCodes/{code} {
      allow get: if signedIn();
      allow list: if false;
      allow create: if isAdmin() && validInviteCode(code) && request.resource.data.keys().hasOnly(['gameId'])
        && getAfter(/databases/$(db)/documents/games/$(request.resource.data.gameId)/secret/invite).data.code == code;
      allow delete: if isAdmin();
      allow update: if false;
    }

    // Configurazione (regolamento e FAQ): tutti leggono, solo l'admin scrive.
    match /config/{id} {
      allow read: if signedIn();
      allow delete: if isAdmin();
      allow create, update: if isAdmin()
        && (id != 'rules'
            || (request.resource.data.keys().hasOnly(['rulesMarkdown', 'faq', 'updatedAt'])
                && request.resource.data.rulesMarkdown is string
                && request.resource.data.rulesMarkdown.size() > 0 && request.resource.data.rulesMarkdown.size() <= 20000
                && request.resource.data.faq is list && request.resource.data.faq.size() <= 30));
    }

    // Chat generale. Solo username (niente contatti), testo max 500, ora del server.
    function chatBanned() { return exists(/databases/$(db)/documents/chatBans/$(request.auth.uid)); }
    match /chat/{id} {
      allow read: if signedIn();
      allow create: if signedIn() && !chatBanned()
        && request.resource.data.keys().hasOnly(['uid', 'username', 'avatar', 'text', 'createdAt', 'isAdmin'])
        && request.resource.data.uid == request.auth.uid
        // etichetta ADMIN: deve corrispondere al ruolo reale
        && request.resource.data.isAdmin == (get(/databases/$(db)/documents/users/$(request.auth.uid)).data.role == 'admin')
        && request.resource.data.username == get(/databases/$(db)/documents/users/$(request.auth.uid)).data.username
        && (!('avatar' in request.resource.data) || (request.resource.data.avatar is string && request.resource.data.avatar.size() <= 16))
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 500
        && request.resource.data.createdAt == request.time
        // limite di frequenza: va aggiornato chatRate/{uid} nello stesso batch
        && getAfter(/databases/$(db)/documents/chatRate/$(request.auth.uid)).data.lastMessageAt == request.time;
      allow delete: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update: if false;
    }

    // Limite: al massimo un messaggio ogni 3 secondi per utente.
    match /chatRate/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;
      allow create, update: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['lastMessageAt'])
        && request.resource.data.lastMessageAt == request.time
        && (resource == null || request.time > resource.data.lastMessageAt + duration.value(3, 's'));
      allow delete: if false;
    }

    // Segnalazioni dei messaggi: le crea chiunque per sé, le legge/elimina solo l'admin.
    match /reports/{id} {
      allow read, delete: if isAdmin();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['messageId', 'reportedUid', 'reportedUsername', 'text', 'reporterUid', 'createdAt'])
        && request.resource.data.reporterUid == request.auth.uid
        && request.resource.data.messageId is string
        && request.resource.data.text is string && request.resource.data.text.size() <= 500
        && request.resource.data.createdAt == request.time;
      allow update: if false;
    }

    // Esclusi dalla chat: ciascuno vede il proprio stato; solo l'admin scrive.
    match /chatBans/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow list: if isAdmin();
      allow write: if isAdmin();
    }

    // Elenco admin dei premi dei vincitori (collection group).
    match /{path=**}/prize/{doc} {
      allow read: if isAdmin();
    }

    // Elenco admin di tutte le richieste di pagamento (collection group).
    match /{path=**}/joinRequests/{uid} {
      allow read: if isAdmin();
    }

    match /games/{gameId} {
      function gamePath() { return /databases/$(db)/documents/games/$(gameId); }
      function game() { return get(gamePath()).data; }
      function isMember() { return exists(/databases/$(db)/documents/games/$(gameId)/players/$(request.auth.uid)); }

      // Le partite (anche private) sono visibili a tutti: il codice invito NON è nel documento.
      allow read: if signedIn();

      // Solo l'admin crea partite, con stato iniziale corretto.
      allow create: if isAdmin()
        && request.resource.data.keys().hasOnly(['name', 'isPublic', 'startMatchday', 'currentMatchday',
             'status', 'createdBy', 'createdAt', 'winnerUid', 'playerCount', 'members', 'startMode', 'targetPlayers',
             'paypalEnabled', 'paypalPrice', 'paypalLink', 'amazonEnabled', 'amazonAmount'])
        && validPayments(request.resource.data)
        && request.resource.data.name is string && request.resource.data.name.size() > 0
        && request.resource.data.isPublic is bool
        && (request.resource.data.isPublic
            || getAfter(/databases/$(db)/documents/games/$(gameId)/secret/invite).data.code is string)
        && request.resource.data.status == 'open'
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.playerCount == 0 && request.resource.data.members == []
        && request.resource.data.winnerUid == null
        && (request.resource.data.get('startMode', 'matchday') == 'players'
          // Avvio al raggiungimento di N iscritti: la giornata la fissa lo script di sync.
          ? (request.resource.data.targetPlayers is int
             && request.resource.data.targetPlayers >= 2 && request.resource.data.targetPlayers <= 500
             && request.resource.data.startMatchday == null && request.resource.data.currentMatchday == null)
          // Avvio da una giornata precisa.
          : (request.resource.data.get('startMode', 'matchday') == 'matchday'
             && request.resource.data.get('targetPlayers', null) == null
             && request.resource.data.startMatchday is int
             && request.resource.data.currentMatchday == request.resource.data.startMatchday
             && exists(/databases/$(db)/documents/matchdays/$(string(request.resource.data.startMatchday)))
             && request.time < get(/databases/$(db)/documents/matchdays/$(string(request.resource.data.startMatchday))).data.deadline));

      // Iscrizione: un utente aggiunge solo sé stesso (members + playerCount), insieme al proprio doc players.
      allow update: if (isAdmin() && validPayments(request.resource.data))
        || (signedIn()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'playerCount'])
          && !(request.auth.uid in resource.data.members)
          && request.resource.data.members == resource.data.members.concat([request.auth.uid])
          && request.resource.data.playerCount == resource.data.playerCount + 1
          && (resource.data.get('startMode', 'matchday') != 'players'
              || resource.data.playerCount < resource.data.targetPlayers)
          && existsAfter(/databases/$(db)/documents/games/$(gameId)/players/$(request.auth.uid)));
      allow delete: if isAdmin();

      // Codice invito della partita privata: solo admin.
      match /secret/{doc} {
        allow read, delete: if isAdmin();
        allow create, update: if isAdmin() && doc == 'invite'
          && request.resource.data.keys().hasOnly(['code'])
          && validInviteCode(request.resource.data.code);
      }

      // Promemoria inviati (scritti dallo script di sync): l'admin li elimina col torneo.
      match /reminders/{md} {
        allow read, delete: if isAdmin();
      }

      match /players/{uid} {
        allow read: if signedIn() && (game().isPublic == true || isMember() || isAdmin());
        // Pubblica: basta che sia aperta. Privata: serve un codice invito esistente che punti a questa partita.
        allow create: if signedIn() && request.auth.uid == uid
          && request.resource.data.keys().hasOnly(['displayName', 'username', 'alive', 'usedTeams', 'joinedAt', 'inviteCode'])
          && request.resource.data.displayName is string
          && request.resource.data.alive == true && request.resource.data.usedTeams == []
          && game().status == 'open'
          && (game().get('startMode', 'matchday') == 'players'
              ? (game().startMatchday == null && game().playerCount < game().targetPlayers)
              : request.time < get(/databases/$(db)/documents/matchdays/$(string(game().startMatchday))).data.deadline)
          && (game().isPublic == true
              || (request.resource.data.inviteCode is string
                  && exists(/databases/$(db)/documents/inviteCodes/$(request.resource.data.inviteCode))
                  && get(/databases/$(db)/documents/inviteCodes/$(request.resource.data.inviteCode)).data.gameId == gameId))
          && uid in getAfter(gamePath()).data.members;
        // L'admin iscrive un utente che ha pagato con PayPal, approvando la sua richiesta nella stessa transazione.
        allow create: if isAdmin()
          && request.resource.data.keys().hasOnly(['displayName', 'username', 'alive', 'usedTeams', 'joinedAt', 'paidRequest'])
          && request.resource.data.displayName is string
          && request.resource.data.alive == true && request.resource.data.usedTeams == []
          && request.resource.data.paidRequest == true
          && game().status == 'open'
          && getAfter(/databases/$(db)/documents/games/$(gameId)/joinRequests/$(uid)).data.status == 'approved'
          && uid in getAfter(gamePath()).data.members;
        allow update: if false;
        allow delete: if isAdmin(); // eliminazione del torneo
      }

      // Richieste di iscrizione a pagamento (PayPal o buono Amazon), confermate a mano dall'admin.
      // Il codice del buono è nella richiesta: la leggono solo l'utente stesso e l'admin.
      match /joinRequests/{uid} {
        function validRequest() {
          return request.resource.data.keys().hasOnly(['uid', 'username', 'displayName', 'amount', 'method', 'code', 'createdAt', 'status'])
            && request.resource.data.uid == uid
            && request.resource.data.displayName is string
            && request.resource.data.status == 'pending'
            && request.resource.data.createdAt == request.time
            && game().isPublic == false
            && (request.resource.data.get('method', 'paypal') == 'amazon'
                ? (game().get('amazonEnabled', false) == true
                   && request.resource.data.amount == game().amazonAmount
                   && request.resource.data.code is string
                   && request.resource.data.code.matches('^[A-Z0-9-]{14,20}$'))
                : (request.resource.data.get('method', 'paypal') == 'paypal'
                   && game().get('paypalEnabled', false) == true
                   && request.resource.data.amount == game().paypalPrice
                   && !('code' in request.resource.data)))
            && game().status == 'open'
            && !(uid in game().members);
        }
        allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
        allow create: if signedIn() && request.auth.uid == uid && validRequest();
        // Dopo un rifiuto l'utente può ripresentare la richiesta.
        allow update: if (signedIn() && request.auth.uid == uid && resource.data.status == 'rejected' && validRequest())
          || (isAdmin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'decidedAt'])
              && request.resource.data.status in ['approved', 'rejected']);
        allow delete: if isAdmin(); // eliminazione del torneo
      }

      // Premio del vincitore: games/{id}/prize/claim. Lo leggono/scrivono solo il vincitore e l'admin.
      // Lo crea lo script di sync a torneo concluso; il vincitore inserisce i dati, l'admin segna "spedito".
      match /prize/{doc} {
        function isWinner() { return signedIn() && game().winnerUid != null && game().winnerUid == request.auth.uid; }
        function str(k, max) { return request.resource.data.get(k, '') is string && request.resource.data.get(k, '').size() <= max; }
        function validClaimData() {
          return str('fullName', 80) && str('street', 120) && str('cap', 5) && str('city', 60)
            && str('province', 2) && str('phone', 20) && str('notes', 500)
            && (request.resource.data.status != 'received'
                || (request.resource.data.fullName.size() > 0 && request.resource.data.street.size() > 0
                    && request.resource.data.cap.matches('^[0-9]{5}$') && request.resource.data.city.size() > 0
                    && request.resource.data.province.matches('^[A-Z]{2}$')
                    && request.resource.data.phone.matches('^[+]?[0-9 ]{6,20}$')));
        }
        function winnerKeys() {
          return ['fullName', 'street', 'cap', 'city', 'province', 'phone', 'notes', 'status', 'submittedAt', 'winnerLastReadAt'];
        }
        allow read: if doc == 'claim' && (isWinner() || isAdmin());
        allow create: if doc == 'claim' && (isAdmin()
          || (isWinner() && request.resource.data.keys().hasOnly(winnerKeys().concat(['winnerUid', 'gameName']))
              && request.resource.data.winnerUid == request.auth.uid
              && request.resource.data.status in ['awaiting', 'received'] && validClaimData()));
        allow update: if doc == 'claim' && (isAdmin()
          || (isWinner() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(winnerKeys())
              && (request.resource.data.status == resource.data.status
                  || (resource.data.status == 'awaiting' && request.resource.data.status == 'received'))
              // dopo la spedizione i dati non si cambiano più
              && (resource.data.status != 'shipped'
                  || request.resource.data.diff(resource.data).affectedKeys().hasOnly(['winnerLastReadAt']))
              && validClaimData()));
        allow delete: if isAdmin();

        // Messaggi privati admin <-> vincitore.
        match /messages/{mid} {
          allow read: if doc == 'claim' && (isWinner() || isAdmin());
          allow create: if doc == 'claim' && (isWinner() || isAdmin())
            && request.resource.data.keys().hasOnly(['uid', 'fromAdmin', 'text', 'createdAt'])
            && request.resource.data.uid == request.auth.uid
            && request.resource.data.fromAdmin == isAdmin()
            && request.resource.data.text is string
            && request.resource.data.text.size() > 0 && request.resource.data.text.size() <= 1000
            && request.resource.data.createdAt == request.time;
          allow update: if false;
          allow delete: if isAdmin();
        }
      }

      // Scelta: id = "{giornata}_{uid}". Modificabile solo prima della scadenza (primo calcio d'inizio).
      match /picks/{pickId} {
        function md() { return request.resource.data.matchday; }
        function deadline() { return get(/databases/$(db)/documents/matchdays/$(string(md()))).data.deadline; }
        function player() { return get(/databases/$(db)/documents/games/$(gameId)/players/$(request.auth.uid)).data; }

        // Le scelte altrui sono visibili solo dopo l'elaborazione della giornata (private: solo agli iscritti).
        allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin()
          || (resource.data.get('processed', false) == true && (game().isPublic == true || isMember())));
        allow create, update: if signedIn()
          && pickId == string(md()) + '_' + request.auth.uid
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.keys().hasOnly(['uid', 'matchday', 'teamId', 'teamName', 'updatedAt'])
          && request.resource.data.teamId is int
          && md() == game().currentMatchday
          && game().status in ['open', 'running']
          && isMember() && player().alive == true
          && !(request.resource.data.teamId in player().usedTeams)
          && deadline() != null && request.time < deadline()
          && (resource == null || resource.data.get('processed', false) == false);
        allow delete: if isAdmin(); // eliminazione del torneo
      }
    }
  }
}
